 CREATE OR REPLACE FUNCTION public.check_rental_block_booking_overlap()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE
  v_category_id uuid;
BEGIN
  SELECT v.category_id INTO v_category_id FROM public.vehicles v WHERE v.id = NEW.vehicle_id;
  IF v_category_id IS NOT NULL THEN
    PERFORM pg_advisory_xact_lock(hashtext('rental_category_' || v_category_id::text));
  END IF;
  PERFORM pg_advisory_xact_lock(hashtext('rental_vehicle_' || NEW.vehicle_id::text));
  IF EXISTS (
    SELECT 1 FROM public.rental_bookings rb
     WHERE rb.vehicle_id = NEW.vehicle_id
       AND rb.status IN ('pending', 'awaiting_payment', 'confirmed', 'active')
       AND rb.booking_period && NEW.block_period
  ) THEN
    RAISE EXCEPTION 'VEHICLE_BOOKING_EXISTS: vehicle % already has an active booking during requested block period', NEW.vehicle_id
      USING ERRCODE = 'RB003';
  END IF;
  RETURN NEW;
END;
$function$

CREATE OR REPLACE FUNCTION public.check_rental_booking_block_overlap()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO 'public', 'pg_temp'
AS $function$
BEGIN
  IF NEW.vehicle_id IS NOT NULL AND NEW.status IN ('pending', 'awaiting_payment', 'confirmed', 'active') THEN
    PERFORM pg_advisory_xact_lock(hashtext('rental_category_' || NEW.category_id::text));
    PERFORM pg_advisory_xact_lock(hashtext('rental_vehicle_' || NEW.vehicle_id::text));
    IF EXISTS (
      SELECT 1 FROM public.rental_blocks b
       WHERE b.vehicle_id = NEW.vehicle_id
         AND b.block_period && NEW.booking_period
    ) THEN
      RAISE EXCEPTION 'VEHICLE_MAINTENANCE_BLOCKED: vehicle % has a maintenance block during requested period', NEW.vehicle_id
        USING ERRCODE = 'RB002';
    END IF;
  END IF;
  RETURN NEW;
END;

CREATE OR REPLACE FUNCTION public.create_rental_booking(p_organizer_id uuid, p_business_id uuid, p_category_id uuid, p_starts_at timestamp with time zone, p_ends_at timestamp with time zone, p_customer_name text, p_customer_email text, p_snapshot_daily_rate numeric, p_rental_amount numeric, p_vehicle_id uuid DEFAULT NULL::uuid, p_source text DEFAULT 'office'::text, p_pickup_branch_id uuid DEFAULT NULL::uuid, p_return_branch_id uuid DEFAULT NULL::uuid, p_customer_phone text DEFAULT NULL::text, p_driver_license_number text DEFAULT NULL::text, p_driver_license_country text DEFAULT NULL::text, p_driver_dob date DEFAULT NULL::date, p_snapshot_weekly_rate numeric DEFAULT NULL::numeric, p_snapshot_currency text DEFAULT 'USD'::text, p_deposit_required_amount numeric DEFAULT 0.00, p_notes text DEFAULT NULL::text, p_expires_at timestamp with time zone DEFAULT NULL::timestamp with time zone, p_booking_number text DEFAULT NULL::text, p_customer_locale text DEFAULT 'en'::text)
 RETURNS rental_bookings
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
#variable_conflict use_column
DECLARE
  v_requested_period   tstzrange;
  v_available_capacity integer;
  v_booking_number     text;
  v_new_booking        public.rental_bookings;
  v_conf_mode          text;
  v_pay_policy         text;
  v_adv_pct            integer;
  v_initial_status     text := 'pending';
  v_amount_due_online  numeric(12,2) := 0.00;
  v_expires_at         timestamptz := p_expires_at;
BEGIN
  IF p_ends_at <= p_starts_at THEN
    RAISE EXCEPTION 'ends_at must be strictly after starts_at' USING ERRCODE = '22000';
  END IF;
  IF NOT EXISTS (
    SELECT 1 FROM public.businesses bz
     WHERE bz.id = p_business_id AND bz.organizer_id = p_organizer_id
  ) THEN
    RAISE EXCEPTION 'TENANT_MISMATCH: business % does not belong to organizer %', p_business_id, p_organizer_id
      USING ERRCODE = 'RB004';
  END IF;
  IF NOT EXISTS (
    SELECT 1 FROM public.vehicle_categories vc
     WHERE vc.id = p_category_id AND vc.business_id = p_business_id AND vc.organizer_id = p_organizer_id
  ) THEN
    RAISE EXCEPTION 'CATEGORY_MISMATCH: category % does not belong to business/organizer', p_category_id
      USING ERRCODE = 'RB005';
  END IF;
  IF p_vehicle_id IS NOT NULL THEN
    IF NOT EXISTS (
      SELECT 1 FROM public.vehicles v
       WHERE v.id = p_vehicle_id
         AND v.category_id = p_category_id
         AND v.business_id = p_business_id
         AND v.organizer_id = p_organizer_id
         AND v.is_active = true
         AND v.status NOT IN ('out_of_service', 'maintenance')
    ) THEN
      RAISE EXCEPTION 'VEHICLE_INVALID: vehicle % does not belong to category/business or is inactive/out_of_service/maintenance', p_vehicle_id
        USING ERRCODE = 'RB006';
    END IF;
  END IF;
  v_requested_period := tstzrange(p_starts_at, p_ends_at, '[)');
  -- Lock order: Category advisory -> Vehicle advisory -> booking rows
  PERFORM pg_advisory_xact_lock(hashtext('rental_category_' || p_category_id::text));
  IF p_vehicle_id IS NOT NULL THEN
    PERFORM pg_advisory_xact_lock(hashtext('rental_vehicle_' || p_vehicle_id::text));
  END IF;
  -- Write-time expiry, limited to this category, skipping open checkout sessions
  UPDATE public.rental_bookings rb
     SET status = 'expired', expired_at = now(), updated_at = now()
   WHERE rb.category_id = p_category_id
     AND rb.status = 'pending'
     AND rb.expires_at IS NOT NULL
     AND rb.expires_at < now()
     AND NOT EXISTS (
       SELECT 1 FROM public.rental_checkout_sessions cs
        WHERE cs.booking_id = rb.id AND cs.status = 'open' AND cs.expires_at > now()
     );
  UPDATE public.rental_bookings rb
     SET status = 'expired', expired_at = now(), updated_at = now()
   WHERE rb.category_id = p_category_id
     AND rb.status = 'awaiting_payment'
     AND rb.payment_due_at IS NOT NULL
     AND rb.payment_due_at < now()
     AND rb.payment_status = 'unpaid'
     AND NOT EXISTS (
       SELECT 1 FROM public.rental_checkout_sessions cs
        WHERE cs.booking_id = rb.id AND cs.status = 'open' AND cs.expires_at > now()
     );
  IF p_vehicle_id IS NOT NULL THEN
    IF EXISTS (
      SELECT 1 FROM public.rental_blocks b
       WHERE b.vehicle_id = p_vehicle_id AND b.block_period && v_requested_period
    ) THEN
      RAISE EXCEPTION 'VEHICLE_MAINTENANCE_BLOCKED: vehicle % has a maintenance block during requested period', p_vehicle_id
        USING ERRCODE = 'RB002';
    END IF;
  END IF;
  WITH eligible_vehicles AS (
    SELECT v.id
      FROM public.vehicles v
     WHERE v.category_id = p_category_id
       AND v.business_id = p_business_id
       AND v.is_active = true
       AND v.status NOT IN ('out_of_service', 'maintenance')
       AND NOT EXISTS (
         SELECT 1 FROM public.rental_blocks b
          WHERE b.vehicle_id = v.id AND b.block_period && v_requested_period
       )
  ),
  conflicting_assigned AS (
    SELECT count(DISTINCT rb.vehicle_id) AS cnt
      FROM public.rental_bookings rb
     WHERE rb.vehicle_id IN (SELECT id FROM eligible_vehicles)
       AND rb.booking_period && v_requested_period
       AND rb.status IN ('pending', 'awaiting_payment', 'confirmed', 'active')
  ),
  conflicting_unassigned AS (
    SELECT count(*) AS cnt
      FROM public.rental_bookings rb
     WHERE rb.category_id = p_category_id
       AND rb.business_id = p_business_id
       AND rb.vehicle_id IS NULL
       AND rb.booking_period && v_requested_period
       AND rb.status IN ('pending', 'awaiting_payment', 'confirmed', 'active')
  )
  SELECT
    (SELECT count(*) FROM eligible_vehicles) -
    (SELECT cnt FROM conflicting_assigned) -
    (SELECT cnt FROM conflicting_unassigned)
  INTO v_available_capacity;
  IF v_available_capacity <= 0 THEN
    RAISE EXCEPTION 'CATEGORY_CAPACITY_EXCEEDED: No available capacity in this category for requested period'
      USING ERRCODE = 'RB001';
  END IF;
  -- Snapshot settings & decide initial status
  SELECT bz.rental_confirmation_mode, bz.rental_payment_policy, bz.rental_advance_pct
    INTO v_conf_mode, v_pay_policy, v_adv_pct
    FROM public.businesses bz
   WHERE bz.id = p_business_id;
  v_conf_mode  := COALESCE(v_conf_mode, 'instant');
  v_pay_policy := COALESCE(v_pay_policy, 'offline');
  IF p_source = 'office' THEN
    v_initial_status    := 'confirmed';
    v_conf_mode         := 'instant';
    v_pay_policy        := 'offline';
    v_adv_pct           := NULL;
    v_amount_due_online := 0.00;
    v_expires_at        := NULL;
  ELSIF v_conf_mode = 'instant' AND v_pay_policy = 'offline' THEN
    v_initial_status    := 'confirmed';
    v_amount_due_online := 0.00;
    v_expires_at        := NULL;
  ELSE
    v_initial_status := 'pending';
    IF v_pay_policy = 'full_online' THEN
      v_amount_due_online := p_rental_amount;
    ELSIF v_pay_policy = 'advance_online' AND v_adv_pct IS NOT NULL THEN
      v_amount_due_online := ROUND(p_rental_amount * v_adv_pct / 100.0, 2);
    ELSE
      v_amount_due_online := 0.00;
    END IF;
    -- Pending bookings must never hold capacity forever
    IF v_expires_at IS NULL THEN
      IF v_conf_mode = 'instant' THEN
        v_expires_at := now() + interval '35 minutes';
      ELSE
        v_expires_at := now() + interval '24 hours';
      END IF;
    END IF;
  END IF;
  IF p_booking_number IS NOT NULL AND p_booking_number <> '' THEN
    v_booking_number := p_booking_number;
  ELSE
    v_booking_number := 'RBK-' || to_char(now() AT TIME ZONE 'UTC', 'YYMMDD') || '-'
                        || lpad(nextval('public.rental_booking_number_seq')::text, 4, '0');
  END IF;
  INSERT INTO public.rental_bookings (
    organizer_id, business_id, category_id, vehicle_id, source, booking_number,
    starts_at, ends_at, expires_at, pickup_branch_id, return_branch_id,
    customer_name, customer_email, customer_phone, customer_locale,
    driver_license_number, driver_license_country, driver_dob,
    status, payment_status, rental_amount, snapshot_daily_rate, snapshot_weekly_rate,
    snapshot_currency, snapshot_confirmation_mode, snapshot_payment_policy,
    snapshot_advance_pct, amount_due_online,
    deposit_required_amount, notes, created_at, updated_at
  ) VALUES (
    p_organizer_id, p_business_id, p_category_id, p_vehicle_id, p_source, v_booking_number,
    p_starts_at, p_ends_at, v_expires_at, p_pickup_branch_id, p_return_branch_id,
    p_customer_name, p_customer_email, p_customer_phone, COALESCE(p_customer_locale, 'en'),
    p_driver_license_number, p_driver_license_country, p_driver_dob,
    v_initial_status, 'unpaid', p_rental_amount, p_snapshot_daily_rate, p_snapshot_weekly_rate,
    p_snapshot_currency, v_conf_mode, v_pay_policy,
    v_adv_pct, v_amount_due_online,
    p_deposit_required_amount, p_notes, now(), now()
  )
  RETURNING * INTO v_new_booking;
  RETURN v_new_booking;
END;
$function$

CREATE OR REPLACE FUNCTION public.get_public_rental_booking(p_token uuid)
 RETURNS TABLE(booking_number text, status text, starts_at timestamp with time zone, ends_at timestamp with time zone, rental_amount numeric, snapshot_currency text, payment_status text, deposit_required_amount numeric, deposit_status text, customer_name text, pickup_branch_id uuid, return_branch_id uuid)
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
BEGIN
  RETURN QUERY
    SELECT
      rb.booking_number,
      rb.status,
      rb.starts_at,
      rb.ends_at,
      rb.rental_amount,
      rb.snapshot_currency,
      rb.payment_status,
      rb.deposit_required_amount,
      rb.deposit_status,
      rb.customer_name,
      rb.pickup_branch_id,
      rb.return_branch_id
    FROM public.rental_bookings rb
    WHERE rb.lookup_token = p_token
    LIMIT 1;
END;

CREATE OR REPLACE FUNCTION public.reconcile_rental_overdue()
 RETURNS integer
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE
  v_count integer;
BEGIN
  UPDATE public.rental_bookings
  SET is_overdue = true,
      updated_at = now()
  WHERE status = 'active'
    AND is_overdue = false
    AND now() > (ends_at + interval '60 minutes');
  GET DIAGNOSTICS v_count = ROW_COUNT;
  RETURN v_count;
END;

CREATE OR REPLACE FUNCTION public.record_product_paid_and_credit(p_order_id uuid, p_stripe_payment_intent_id text, p_crypto_payment_id text)
 RETURNS TABLE(order_id uuid, product_id uuid, quantity integer, was_newly_paid boolean)
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_updated_id UUID;
  v_product_id UUID;
  v_quantity INTEGER;
  v_currency TEXT;
  v_total_amount NUMERIC;
  v_owner_id UUID;
  v_recipient_id UUID;
BEGIN
  UPDATE product_orders
  SET
    status = 'paid',
    stripe_payment_intent_id = COALESCE(p_stripe_payment_intent_id, stripe_payment_intent_id),
    crypto_payment_id        = COALESCE(p_crypto_payment_id, crypto_payment_id)
  WHERE id = p_order_id
    AND status = 'pending'
  RETURNING
    id, product_orders.product_id, product_orders.quantity,
    product_orders.currency, product_orders.total_amount
  INTO v_updated_id, v_product_id, v_quantity, v_currency, v_total_amount;
  IF v_updated_id IS NULL THEN
    SELECT product_orders.product_id, product_orders.quantity
    INTO v_product_id, v_quantity
    FROM product_orders
    WHERE id = p_order_id;
    RETURN QUERY SELECT p_order_id, v_product_id, v_quantity, false;
    RETURN;
  END IF;
  SELECT owner_id INTO v_owner_id
  FROM products
  WHERE id = v_product_id;
  IF v_owner_id IS NULL THEN
    RAISE EXCEPTION
      'record_product_paid_and_credit: product % not found or has no owner -- cannot resolve recipient',
      v_product_id;
  END IF;
  v_recipient_id := resolve_recipient('user', v_owner_id, NULL, NULL);
  INSERT INTO recipient_ledger_entries (
    recipient_id,
    entry_type,
    amount,
    currency,
    source_type,
    source_id,
    description
  ) VALUES (
    v_recipient_id,
    'credit',
    v_total_amount,
    lower(v_currency),
    'product_order',
    v_updated_id,
    'Product order received'
  );
  RETURN QUERY SELECT v_updated_id, v_product_id, v_quantity, true;
END;
$function$

CREATE OR REPLACE FUNCTION public.record_ticket_and_credit(p_event_id uuid, p_ticket_id uuid, p_seat_id uuid, p_seat_label text, p_buyer_email text, p_buyer_name text, p_quantity integer, p_total_amount numeric, p_currency text, p_qr_code text DEFAULT NULL::text, p_stripe_payment_intent_id text DEFAULT NULL::text, p_stripe_session_id text DEFAULT NULL::text, p_items_json jsonb DEFAULT NULL::jsonb, p_seats_json jsonb DEFAULT NULL::jsonb)
 RETURNS TABLE(ticket_order_id uuid, is_new boolean)
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_ticket_order_id UUID;
  v_event_organizer_id UUID;
  v_event_user_id UUID;
  v_recipient_id UUID;
  v_total_qty INTEGER := 0;
  v_item JSONB;
  v_item_ticket_id UUID;
  v_item_qty INTEGER;
  v_seat JSONB;
  v_seat_ticket_id UUID;
  v_seat_id UUID;
  v_seat_label TEXT;
  v_first_seat JSONB;
  v_primary_ticket_id UUID;
  v_primary_seat_id UUID;
  v_primary_seat_label TEXT;
  i INTEGER;
BEGIN
  -- Idempotency check: if payment_intent_id already exists, return existing order_id and false
  IF p_stripe_payment_intent_id IS NOT NULL THEN
    SELECT id INTO v_ticket_order_id
    FROM ticket_orders
    WHERE stripe_payment_intent_id = p_stripe_payment_intent_id;
    IF v_ticket_order_id IS NOT NULL THEN
      RETURN QUERY SELECT v_ticket_order_id, false;
      RETURN;
    END IF;
  END IF;
  -- Idempotency check: if stripe_session_id already exists
  IF p_stripe_session_id IS NOT NULL THEN
    SELECT id INTO v_ticket_order_id
    FROM ticket_orders
    WHERE stripe_session_id = p_stripe_session_id;
    IF v_ticket_order_id IS NOT NULL THEN
      RETURN QUERY SELECT v_ticket_order_id, false;
      RETURN;
    END IF;
  END IF;
  -- Calculate total quantity
  IF p_seats_json IS NOT NULL AND jsonb_array_length(p_seats_json) > 0 THEN
    v_total_qty := jsonb_array_length(p_seats_json);
  ELSIF p_items_json IS NOT NULL AND jsonb_array_length(p_items_json) > 0 THEN
    FOR v_item IN SELECT * FROM jsonb_array_elements(p_items_json) LOOP
      v_total_qty := v_total_qty + COALESCE((v_item->>'quantity')::INTEGER, 1);
    END LOOP;
  ELSE
    v_total_qty := GREATEST(1, COALESCE(p_quantity, 1));
  END IF;
  -- Derive primary ticket_id, seat_id, seat_label
  IF p_seats_json IS NOT NULL AND jsonb_array_length(p_seats_json) > 0 THEN
    v_first_seat := p_seats_json->0;
    v_primary_ticket_id := COALESCE(
      p_ticket_id,
      NULLIF(v_first_seat->>'ticket_id', '')::UUID,
      NULLIF(v_first_seat->>'t', '')::UUID
    );
    v_primary_seat_id := COALESCE(
      p_seat_id,
      NULLIF(v_first_seat->>'seat_id', '')::UUID,
      NULLIF(v_first_seat->>'id', '')::UUID,
      NULLIF(v_first_seat->>'s', '')::UUID
    );
    v_primary_seat_label := COALESCE(
      p_seat_label,
      v_first_seat->>'seat_label',
      v_first_seat->>'label',
      v_first_seat->>'l'
    );
  ELSE
    v_primary_ticket_id := p_ticket_id;
    v_primary_seat_id := p_seat_id;
    v_primary_seat_label := p_seat_label;
  END IF;
  -- Insert 1 summary ticket_orders row
  INSERT INTO ticket_orders (
    event_id,
    ticket_id,
    seat_id,
    seat_label,
    buyer_email,
    buyer_name,
    quantity,
    total_amount,
    currency,
    qr_code,
    status,
    stripe_payment_intent_id,
    stripe_session_id
  ) VALUES (
    p_event_id,
    v_primary_ticket_id,
    v_primary_seat_id,
    v_primary_seat_label,
    p_buyer_email,
    p_buyer_name,
    v_total_qty,
    p_total_amount,
    p_currency,
    p_qr_code,
    'valid',
    p_stripe_payment_intent_id,
    p_stripe_session_id
  )
  RETURNING id INTO v_ticket_order_id;
  -- Insert ticket_instances:
  -- Path A: Multi-seat order (p_seats_json provided)
  IF p_seats_json IS NOT NULL AND jsonb_array_length(p_seats_json) > 0 THEN
    FOR v_seat IN SELECT * FROM jsonb_array_elements(p_seats_json) LOOP
      v_seat_ticket_id := COALESCE(
        NULLIF(v_seat->>'ticket_id', '')::UUID,
        NULLIF(v_seat->>'t', '')::UUID,
        p_ticket_id
      );
      v_seat_id := COALESCE(
        NULLIF(v_seat->>'seat_id', '')::UUID,
        NULLIF(v_seat->>'id', '')::UUID,
        NULLIF(v_seat->>'s', '')::UUID
      );
      v_seat_label := COALESCE(
        v_seat->>'seat_label',
        v_seat->>'label',
        v_seat->>'l'
      );
      -- Safety guard: Stale-hold double-sell race condition check.
      -- If another order already holds an active (non-cancelled, non-refunded) instance
      -- for this seat, abort immediately so the webhook reconciliation alert is triggered.
      IF v_seat_id IS NOT NULL THEN
        IF EXISTS (
          SELECT 1
          FROM ticket_instances
          WHERE seat_id = v_seat_id
            AND event_id = p_event_id
            AND order_id <> v_ticket_order_id
            AND status NOT IN ('cancelled', 'refunded')
        ) THEN
          RAISE EXCEPTION 'SEAT_ALREADY_ASSIGNED: seat % already sold to a different order', v_seat_id;
        END IF;
      END IF;
      INSERT INTO ticket_instances (
        order_id,
        event_id,
        ticket_id,
        seat_id,
        seat_label,
        qr_code,
        status
      ) VALUES (
        v_ticket_order_id,
        p_event_id,
        v_seat_ticket_id,
        v_seat_id,
        v_seat_label,
        UPPER(REPLACE(GEN_RANDOM_UUID()::text, '-', '')),
        'valid'
      );
    END LOOP;
    -- Update seats to sold
    UPDATE seats
    SET status = 'sold', reserved_until = NULL
    WHERE id IN (
      SELECT COALESCE(
        NULLIF(s->>'seat_id', '')::UUID,
        NULLIF(s->>'id', '')::UUID,
        NULLIF(s->>'s', '')::UUID
      )
      FROM jsonb_array_elements(p_seats_json) s
      WHERE COALESCE(
        NULLIF(s->>'seat_id', '')::UUID,
        NULLIF(s->>'id', '')::UUID,
        NULLIF(s->>'s', '')::UUID
      ) IS NOT NULL
    );
  -- Path B: Non-seat multi-tier order (p_items_json provided)
  ELSIF p_items_json IS NOT NULL AND jsonb_array_length(p_items_json) > 0 THEN
    FOR v_item IN SELECT * FROM jsonb_array_elements(p_items_json) LOOP
      v_item_ticket_id := (v_item->>'ticket_id')::UUID;
      v_item_qty := COALESCE((v_item->>'quantity')::INTEGER, 1);
      FOR i IN 1..v_item_qty LOOP
        INSERT INTO ticket_instances (
          order_id,
          event_id,
          ticket_id,
          seat_id,
          seat_label,
          qr_code,
          status
        ) VALUES (
          v_ticket_order_id,
          p_event_id,
          v_item_ticket_id,
          p_seat_id,
          p_seat_label,
          UPPER(REPLACE(GEN_RANDOM_UUID()::text, '-', '')),
          'valid'
        );
      END LOOP;
    END LOOP;
    IF p_seat_id IS NOT NULL THEN
      UPDATE seats
      SET status = 'sold', reserved_until = NULL
      WHERE id = p_seat_id;
    END IF;
  -- Path C: Legacy single-tier order
  ELSE
    FOR i IN 1..v_total_qty LOOP
      INSERT INTO ticket_instances (
        order_id,
        event_id,
        ticket_id,
        seat_id,
        seat_label,
        qr_code,
        status
      ) VALUES (
        v_ticket_order_id,
        p_event_id,
        p_ticket_id,
        p_seat_id,
        p_seat_label,
        UPPER(REPLACE(GEN_RANDOM_UUID()::text, '-', '')),
        'valid'
      );
    END LOOP;
    IF p_seat_id IS NOT NULL THEN
      UPDATE seats
      SET status = 'sold', reserved_until = NULL
      WHERE id = p_seat_id;
    END IF;
  END IF;
  -- Recipient resolution & single ledger credit
  SELECT organizer_id, user_id
  INTO v_event_organizer_id, v_event_user_id
  FROM events
  WHERE id = p_event_id;
  IF v_event_organizer_id IS NOT NULL THEN
    v_recipient_id := resolve_recipient('organizer', NULL, v_event_organizer_id, NULL);
  ELSIF v_event_user_id IS NOT NULL THEN
    v_recipient_id := resolve_recipient('user', v_event_user_id, NULL, NULL);
  ELSE
    RAISE EXCEPTION
      'record_ticket_and_credit: event % has no organizer_id or user_id -- cannot resolve recipient',
      p_event_id;
  END IF;
  INSERT INTO recipient_ledger_entries (
    recipient_id,
    entry_type,
    amount,
    currency,
    source_type,
    source_id,
    description
  ) VALUES (
    v_recipient_id,
    'credit',
    p_total_amount,
    lower(p_currency),
    'ticket_order',
    v_ticket_order_id,
    'Ticket order received'
  );
  RETURN QUERY SELECT v_ticket_order_id, true;
END;
$function$

CREATE OR REPLACE FUNCTION public.resolve_recipient(p_recipient_type text, p_user_id uuid DEFAULT NULL::uuid, p_organizer_id uuid DEFAULT NULL::uuid, p_business_id uuid DEFAULT NULL::uuid)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  result_id UUID;
BEGIN
  IF p_recipient_type = 'user' THEN
    IF p_user_id IS NULL OR p_organizer_id IS NOT NULL OR p_business_id IS NOT NULL THEN
      RAISE EXCEPTION 'resolve_recipient: user recipient requires user_id only';
    END IF;
    INSERT INTO recipients (recipient_type, user_id)
    VALUES ('user', p_user_id)
    ON CONFLICT (user_id) WHERE user_id IS NOT NULL DO NOTHING
    RETURNING id INTO result_id;
    IF result_id IS NULL THEN
      SELECT id INTO result_id FROM recipients WHERE user_id = p_user_id;
    END IF;
  ELSIF p_recipient_type = 'organizer' THEN
    IF p_organizer_id IS NULL OR p_user_id IS NOT NULL OR p_business_id IS NOT NULL THEN
      RAISE EXCEPTION 'resolve_recipient: organizer recipient requires organizer_id only';
    END IF;
    INSERT INTO recipients (recipient_type, organizer_id)
    VALUES ('organizer', p_organizer_id)
    ON CONFLICT (organizer_id) WHERE organizer_id IS NOT NULL DO NOTHING
    RETURNING id INTO result_id;
    IF result_id IS NULL THEN
      SELECT id INTO result_id FROM recipients WHERE organizer_id = p_organizer_id;
    END IF;
  ELSIF p_recipient_type = 'business' THEN
    IF p_business_id IS NULL OR p_user_id IS NOT NULL OR p_organizer_id IS NOT NULL THEN
      RAISE EXCEPTION 'resolve_recipient: business recipient requires business_id only';
    END IF;
    INSERT INTO recipients (recipient_type, business_id)
    VALUES ('business', p_business_id)
    ON CONFLICT (business_id) WHERE business_id IS NOT NULL DO NOTHING
    RETURNING id INTO result_id;
    IF result_id IS NULL THEN
      SELECT id INTO result_id FROM recipients WHERE business_id = p_business_id;
    END IF;
  ELSE
    RAISE EXCEPTION 'resolve_recipient: unknown recipient_type %', p_recipient_type;
  END IF;
  RETURN result_id;
END;
$function$
